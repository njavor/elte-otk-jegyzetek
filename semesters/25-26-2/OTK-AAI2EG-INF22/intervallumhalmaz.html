    public IntervallumHalmaz Komplementer(int k, int v)
    {
        IntervallumHalmaz komplementer = new IntervallumHalmaz();
        if (db == 0)
        {
            komplementer.Halmazba(1, 10);
        }
        else
        {
            if (kezd[0] > k) // rendezvény eleje - első foglalás
            {
                komplementer.Halmazba(k, kezd[0] - 1);
            }
            for (int i = 0; i < db - 1; i++) // foglalás[i] - foglalás[i+1]
            {
                if (kezd[i + 1] - veg[i] > 1)
                {
                    komplementer.Halmazba(veg[i] + 1, kezd[i + 1] - 1);
                }
            }
            if (veg[db - 1] < v) // utolsó foglalás - rendezvény vége
            {
                komplementer.Halmazba(veg[db - 1] + 1, v);
            }
        }
        return komplementer;
    }
