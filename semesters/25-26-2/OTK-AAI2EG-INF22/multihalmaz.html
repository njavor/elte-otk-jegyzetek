A multihalmaz

A multihalmaz egy olyan halmaz, ami értékeken túl multiplicitást (darabszámot) is tárol. Például,

van egy allatok = {"kenguru", "koala", "krokodil", "kecske"} halmazom, és egy allatokMulti = {("kenguru", 4), ("koala", 2), ("krokodil", 3), ("kecske", 24)} multihalmazom. Az allatokMultiban azon túl, hogy látom, hogy milyen állataim vannak, azt is látom, hogy ezekből az állatokból hány db van. Ha hoznak még kettő koalát hozzám, akkor az allatok halmaz nem módosulna, viszont az allatokMulti ("koala", 2) értékéből ("koala", 4) lenne.

        MultihalmazElem[] elemek;

        {
            db = 0;
            elemek = new MultihalmazElem[MaxDb];
        }

    }

Beolvasás és kiírás

Beolvasás


    public void Beolvasas()
    {
        db = int.Parse(Console.ReadLine());
        for (int i = 0; i < db; i++)
        {
            string[] sor = Console.ReadLine().Split();
            elemek[i] = new MultihalmazElem(int.Parse(sor[0]), int.Parse(sor[1]));
        }
    }

Kiírás


    public void Kiiras()
    {
        Console.WriteLine(db);
        for (int i = 0; i < db; i++)
        {
            Console.Write(elemek[i] + " "); // akkor működik, ha a MultihalmazElem ToString()-je felül van írva
        }
        Console.WriteLine();
    }

Multihalmaz módosítása

Multihalmazba


    public void Multihalmazba(MultihalmazElem elem)
    {
        int i = 0;
        while (i < db && elemek[i].ertek != elem.ertek) i++;
        if (i < db) elemek[i].multi++;
        else
        {
            elemek[db] = elem;
            db++;
        }
    }

Multihalmazból


    public void Multihalmazbol(MultihalmazElem elem)
    {
        int i = 0;
        while (i < db && elemek[i].ertek != elem.ertek) i++;
        if (i < db)
        {
            elemek[i].multi -= elem.multi;
            if (elemek[i].multi <= 0)
            {
                elemek[i] = elemek[db - 1];
                db--;
            }
        }
    }

Ürítés


    public void Urites()
    {
        db = 0;
    }

Logikai vizsgálatok

Eleme-e

Az elemérték benne van-e a multihalmazban


    public bool ElemeE(int ertek)
    {
        int i = 0;
        while (i < db && elemek[i].ertek != ertek) i++;
        return i < db;
    }

Benne-e

Az elem benne van-e a multihalmazban.


    public bool BenneE(MultihalmazElem elem)
    {
        int i = 0;
        while (i < db && elemek[i].ertek != elem.ertek) i++;
        return i < db && elemek[i].multi >= elem.multi;
    }

Üres-e

0 elemet tartalmaz-e a multihalmaz.


    public bool UresE()
    {
        return db == 0;
    }

Része-e

A multihalmaz benne van-e a másik multihalmazban.


    public bool ReszeE(Multihalmaz masik)
    {
        if (masik.db < this.db) return false;

        int i = 0;
        while (i < db && masik.BenneE(elemek[i])) i++;
        return i >= db;
    }

Halmazműveletek

Unió


    public Multihalmaz Unio(Multihalmaz masik)
    {
        Multihalmaz unio = new Multihalmaz();
        for (int i = 0; i < this.db; i++) unio.Multihalmazba(this.elemek[i]);
        for (int i = 0; i < masik.db; i++) unio.Multihalmazba(masik.elemek[i]);
        return unio;
    }

Metszet


    public Multihalmaz Metszet(Multihalmaz masik)
    {
        Multihalmaz metszet = new Multihalmaz();
        for (int i = 0; i < this.db; i++)
        {
            if (masik.BenneE(elemek[i])) metszet.Multihalmazba(elemek[i]);
        }
        return metszet;
    }

Különbség


    public Multihalmaz Kulonbseg(Multihalmaz masik)
    {
        Multihalmaz kulonbseg = new Multihalmaz();
        for (int i = 0; i < this.db; i++)
        {
            if (!masik.BenneE(elemek[i])) kulonbseg.Multihalmazba(elemek[i]);
        }
        return kulonbseg;
    }
